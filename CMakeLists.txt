cmake_minimum_required(VERSION 3.22)
project(stimulus VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(STIMULUS_BUILD_APP "Build the graphical application" ON)
set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_PKGCONFIG_SUPPORT OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_OBJECT_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(jsoncpp GIT_REPOSITORY https://github.com/open-source-parsers/jsoncpp.git GIT_TAG 89e2973c754a9c02a49974d839779b151e95afd6)
FetchContent_MakeAvailable(jsoncpp)
add_library(stimulus_core src/model.cpp src/storage.cpp)
target_include_directories(stimulus_core PUBLIC src)
target_link_libraries(stimulus_core PUBLIC jsoncpp_static)
target_compile_options(stimulus_core PRIVATE -Wall -Wextra -Wpedantic)
if(EMSCRIPTEN)
  target_compile_options(jsoncpp_static PRIVATE -fexceptions)
  target_compile_options(stimulus_core PUBLIC -fexceptions)
  target_link_options(stimulus_core PUBLIC -fexceptions)
endif()
if(STIMULUS_BUILD_APP)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  if(EMSCRIPTEN)
    set(PLATFORM Web CACHE STRING "" FORCE)
  endif()
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
  FetchContent_MakeAvailable(raylib)
  add_executable(stimulus src/main.cpp src/app.cpp src/platform.cpp)
  target_link_libraries(stimulus PRIVATE stimulus_core raylib)
  target_compile_options(stimulus PRIVATE -Wall -Wextra -Wpedantic)
  if(EMSCRIPTEN)
    set_target_properties(stimulus PROPERTIES OUTPUT_NAME index SUFFIX ".html" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web")
    target_link_options(stimulus PRIVATE "--shell-file=${CMAKE_SOURCE_DIR}/web/shell.html" -sUSE_GLFW=3 "-sEXPORTED_RUNTIME_METHODS=['ccall']" -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sEXIT_RUNTIME=0)
    set_property(TARGET stimulus APPEND PROPERTY LINK_DEPENDS "${CMAKE_SOURCE_DIR}/web/shell.html")
  endif()
endif()
include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE stimulus_core)
  add_test(NAME core COMMAND core_tests)
endif()
